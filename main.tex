% --- DEFINICIÓN DE ESTILOS PERSONALIZADOS ---

% 1. Estilo para Proposiciones (Título Azul, Texto en Cursiva)
\newtheoremstyle{estilo-proposicion-azul}
  {10pt}        % Espacio arriba
  {10pt}        % Espacio abajo
  {\itshape}    % Fuente del cuerpo (Cursiva para matemáticas)
  {}            % Indentación
  {\bfseries\color{blue}} % Fuente del título (Negrita y AZUL)
  {.}           % Puntuación después del título
  {0.5em}       % Espacio después del título
  {\thmname{#1} \thmnumber{#2}\thmnote{ (#3)}}            % Especificación manual (dejar vacío)

% 2. Estilo para Ejemplos/Ejercicios (Título Azul, Texto Normal)
\newtheoremstyle{estilo-ejemplo-azul}
  {10pt}        % Espacio arriba
  {10pt}        % Espacio abajo
  {\upshape}    % Fuente del cuerpo (Texto normal, no cursiva)
  {}            % Indentación
  {\bfseries\color{blue}} % Fuente del título (Negrita y AZUL)
  {.}           % Puntuación después del título
  {0.5em}       % Espacio después del título
  {\thmname{#1} \thmnumber{#2}\thmnote{ (#3)}}           % Especificación manual

% --- ASIGNACIÓN DE ENTORNOS ---

\newtheoremstyle{estilo-nota-azul}
  {10pt}        % Espacio arriba
  {10pt}        % Espacio abajo
  {\upshape}    % Fuente del cuerpo (Texto normal, no cursiva)
  {}            % Indentación
  {\itshape\color{blue}} % Fuente del título (Cursiva y AZUL)
  {.}           % Puntuación después del título
  {0.5em}       % Espacio después del título
  {\thmname{#1} \thmnumber{#2}\thmnote{ (#3)}} % Orden exacto: Nombre + Número + (Nota)
